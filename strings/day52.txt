#checkAcronym

class Solution {
public:
    bool isAcronym(vector<string>& words, string s) {
        string acro = "";
        for (int i = 0; i < words.size(); i++)
        {
            string word = words[i];
            acro += word[0];
        }
        if (acro != s)
        {
            return false;
        }
        return true;
    }
};

class Solution {
public:
    bool isAcronym(vector<string>& words, string s) {
        if (s.length() != words.size())
        {
            return false;
        }
        for (int i = 0; i < words.size(); i++)
        {
            string w = words[i];
            if (s[i] != w[0])
            {
                return false;
            }
        }
        return true;
    }
};
